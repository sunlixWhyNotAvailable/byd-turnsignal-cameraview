cmake_minimum_required(VERSION 3.22.1)
project(motion_yuv LANGUAGES CXX)
add_library(motion_yuv SHARED motion_yuv.cpp motion_hardware.cpp)
target_link_libraries(motion_yuv android EGL GLESv2 log dl)
target_compile_features(motion_yuv PRIVATE cxx_std_17)
target_compile_options(motion_yuv PRIVATE -O3 -Wall -Wextra -Werror)

add_library(camera_input_queue SHARED camera_input_queue.cpp)
target_link_libraries(camera_input_queue android log dl)
target_compile_features(camera_input_queue PRIVATE cxx_std_17)
target_compile_options(camera_input_queue PRIVATE -O2 -Wall -Wextra -Werror)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    target_compile_options(camera_input_queue PRIVATE -mbranch-protection=standard)
endif()
